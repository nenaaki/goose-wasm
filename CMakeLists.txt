cmake_minimum_required(VERSION 3.24)
project(goose_wasm C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Upstream Goose, unmodified (git submodule).
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/upstream/goose")
if(NOT EXISTS "${UPSTREAM}/src/main.cpp")
    message(FATAL_ERROR "upstream/goose is missing: git submodule update --init")
endif()

# goose_version.h for upstream's src/version.cpp, written by upstream's own
# script from `git describe` of the submodule.
set(version_h "${CMAKE_CURRENT_BINARY_DIR}/version/goose_version.h")
add_custom_target(goose_version
    COMMAND "${CMAKE_COMMAND}" "-DSRC=${UPSTREAM}" "-DOUT=${version_h}"
            -P "${UPSTREAM}/cmake/version.cmake"
    BYPRODUCTS "${version_h}")

# builtin_packages.h, which upstream's builtin.h includes: the blessed table
# and no packages compiled in. Upstream's goose_builtin_packages
# (cmake/package.cmake) writes it for its own `goose` target and tree, so
# this writes the same form, from upstream's src/blessed.txt.
file(READ "${UPSTREAM}/src/blessed.txt" blessed)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/builtin_packages.h" CONTENT
"// Generated by goose-wasm's CMakeLists.txt in the form upstream's
// cmake/package.cmake writes: the blessed table, no packages compiled in.
#pragma once

namespace goose {

inline const char *blessed_table = R\"GSBL(${blessed})GSBL\";

inline const BuiltinPackage builtin_packages[] = {
    { nullptr, nullptr, nullptr, nullptr },
};

}  // namespace goose
" @ONLY)

# goose-wasm's replacements for standard library files whose extern fns
# upstream's C runtime implements (lib/README.md): compiled in, and served
# by the driver in place of upstream's file of the same name.
set(lib_overrides slice_pool_internal.goose math.goose)
set(overrides_body "")
foreach(name IN LISTS lib_overrides)
    set(src "${CMAKE_CURRENT_SOURCE_DIR}/lib/${name}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${src}")
    file(READ "${src}" text)
    string(APPEND overrides_body "    { \"${name}\", R\"GSWL(${text})GSWL\" },
")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/lib_overrides.h" CONTENT
"// Generated by goose-wasm's CMakeLists.txt from lib/: standard library
// files the driver serves in place of upstream's.
#pragma once

namespace goosewasm {

struct LibOverride { const char *name, *text; };
inline const LibOverride lib_overrides[] = {
${overrides_body}};

}  // namespace goosewasm
" @ONLY)

# The shader compiler behind embed_shader, as upstream builds it
# (third_party/cute_spirv: zlib or public domain).
add_library(goose_shaderc STATIC "${UPSTREAM}/src/shaderc.c")
target_include_directories(goose_shaderc PRIVATE "${UPSTREAM}/third_party/cute_spirv")
set_target_properties(goose_shaderc PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)
target_compile_options(goose_shaderc PRIVATE -w)

add_executable(goose-wasm
    src/driver.cpp
    "${UPSTREAM}/src/process.cpp"
    "${UPSTREAM}/src/progress.cpp"
    "${UPSTREAM}/src/version.cpp")
add_dependencies(goose-wasm goose_version)
target_include_directories(goose-wasm PRIVATE
    src
    "${UPSTREAM}/src"
    "${UPSTREAM}/third_party/subprocess"
    "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${CMAKE_CURRENT_BINARY_DIR}/version")
# Upstream's standard library, unless --stdlib or GOOSE_STDLIB names another.
target_compile_definitions(goose-wasm PRIVATE
    "GOOSE_WASM_DEFAULT_STDLIB=\"${UPSTREAM}/stdlib\"")
find_package(Threads REQUIRED)
target_link_libraries(goose-wasm PRIVATE goose_shaderc Threads::Threads)

# The typechecker recurses as deep as a program's call paths. Upstream runs
# the compile on a thread with a 64 MB stack; this driver runs it on the main
# thread, given the same 64 MB. Static runtime libraries, as upstream's MSVC
# build links them, so the compiler starts with nothing on PATH.
if(MINGW)
    target_link_options(goose-wasm PRIVATE "LINKER:--stack,67108864" -static)
elseif(APPLE)
    target_link_options(goose-wasm PRIVATE "LINKER:-stack_size,0x4000000")
endif()
